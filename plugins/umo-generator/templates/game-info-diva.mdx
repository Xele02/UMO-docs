export const DivaStatsRow = ({data, prevData, idx}) =>
{
  const divaId = useContext(DivaIdContext_);
  const list = ["ANAJIAENLNB", "HFIDCMNFBJG", "PFJCOCPKABN", "JFJDLEMNKFE", "GDOLPGBLMEA", "ONDFNOOICLE", "HCFOMFDPGEC"]
  const col = list.map(c => {
    const prevVal = prevData?.[c];
    const val = data[c];
    const _path = "CDENCMNHNGA/"+(divaId - 1)+"/CMCKNKKCNDK/"+idx+"/"+c;
    return <td><DatabaseValue database_name="diva" path={_path} />{(c != "ANAJIAENLNB" && prevVal && prevVal != val) ? " (+"+(val-prevVal)+")" : ""}</td>
  });
  return (<><tr>
    {col}
  </tr></>);
}

export const DivaStatsRows = ({values_list}) =>
{
  const divaId = useContext(DivaIdContext_);

  const data = getPageData();
  const reportError = useReportError();

  const data_db = data?.database?.diva;

  const valueDb = getDatabaseValueFromData(data_db, "AGNCAAFGLBE", (d) =>
  {
    useEffect(() => {
        reportError('Missing database value '+d[0]+' in path data.database.diva.'+d[1]);
    }, [data, reportError]);
  });
  useEffect(() => {
    if(!data_db?.CDENCMNHNGA?.[(divaId - 1)]?.CMCKNKKCNDK)
      reportError('Missing diva datas data.database.diva.CDENCMNHNGA.['+(divaId-1)+'].CMCKNKKCNDK');
  }, [data, reportError]);
  const maxLevels = parseInt(valueDb || 0);

  const stats = data_db?.CDENCMNHNGA?.[(divaId - 1)]?.CMCKNKKCNDK || {};
  var prev = undefined;
  return Object.entries(stats).sort(([keyA], [keyB]) => Number(keyA) - Number(keyB)).map(([idx, k]) => {
    const _p = prev;
    const lvl = parseInt(k.ANAJIAENLNB);
    if(!values_list || values_list.includes(lvl))
      prev = k;
    if(lvl <= maxLevels && (!values_list || values_list.includes(lvl)))
      return <DivaStatsRow data={k} prevData={_p} idx={idx} />
    else
      return null;
  });
}

export const CostumeClick = ({divaId}) => {

    const data = getPageData();
    const reportError = useReportError();
    const cos = Object.entries(getDatabaseValueFromData(data?.database?.costume, "CDENCMNHNGA", (d) =>
    {
      useEffect(() => {
        reportError('Missing database value '+d[0]+' in path data.database.costume.'+d[1]);
    }, [data, reportError]);
    }))?.filter(([idx, c]) => c.AHHJLDLAPAN == divaId && c.PPEGAKEIEGM == 2) || [];
    const list = cos.sort(([keyA], [keyB]) => Number(keyA) - Number(keyB)).map(([idx, k]) => {
      const id = k.JPIDIENBGKH;
      const hasColor = k._has_color;
      const url = getLinkById("documentation_game_costume_"+id.toString());
      const colorImg = hasColor && (<><span style={{position:'absolute', maxWidth:'50%', right:'0', bottom:'0', backgroundColor:'#FFBBBBAA', borderRadius:'50%'}}><CostumeImage type="costume" costumeId={{id:id}} colorId="1" /></span></>);
      const colorName = hasColor ? (<>/ <CostumeName costumeId={{id:id}} colorId="1" /></>) : "";
      return (
          <article class="col col--2 margin-bottom--lg"><Link
          href={url}
          className={clsx('card ', styles.cardContainer)}>
          <div style={{position:'relative'}}><CostumeImage type="costume" costumeId={{id:id}} />
          {colorImg}</div>
          <CostumeName costumeId={{id:id}} />{colorName}
          </Link></article>
      );
  });
  return <>{list}</>;
}
